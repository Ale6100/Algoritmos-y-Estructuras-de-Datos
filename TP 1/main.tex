    \auxLargo{millasTotales}{s: AirMiles, u: Usuario}{\R}{sumaDeAcumulaciones(s,u) + sumaDeTransferenciasRecibidas(s,u) + sumaReferidos(s,u)}
    
    \bigskip
    
    \auxLargo{millasDisponibles}{s: AirMiles, u: Usuario}{\R}{millasTotales(s,u) - sumaDeCanjes(s,u) - sumaDeTransferenciasEnviadas(s,u)}
    
    \bigskip
    
    \auxLargo{sumaDeAcumulaciones}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.historialDeAcumulaciones[u]|-1}{s.historialDeAcumulaciones[u][i].millasGanadas}}
    
    
    \bigskip
    
    \auxLargo{sumaDeTransferenciasRecibidas}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.historialDeTransferencias|-1}
        \operatorname{IfThenElse}\big(
        s.historialDeTransferencias[i].destino = u, s.historialDeTransferencias[i].cantidad,0\big)}
    
    \bigskip
    
    \auxLargo{sumaDeTransferenciasEnviadas}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.historialDeTransferencias|-1}
        \operatorname{IfThenElse}\big(
        s.historialDeTransferencias[i].origen = u, s.historialDeTransferencias[i].cantidad,0\big)}
    
    \bigskip
    
    \aux{sumaReferidos}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.referidos[u]|-1}{s.referidos[u][i]}}

    \bigskip

    \auxLargo{sumaDeCanjes}{u: \Z, s: \Tipo{AirMiles}}{\R}{\sum_{i=0}^{|s.historialDeCanjes[u]|-1}
    s.historialDeCanjes[u][i]}
    
    \bigskip
    
    \auxLargo{categoria}{s: AirMiles, u:Usuario}{string}{(millasTotales(s,u) < 10.000 \implica res = "Bronce") \lor \\ (10.000 \leq millasTotales(s,u) < 50.000 \implica res = "Plata") \lor \\ (50.000 \leq millasTotales(s,u) < 1.000.000 \implica res = "Oro") \lor \\ (millasTotales (s,u) \geq 1.000.000 \implies res = "Platino")}
    
    \bigskip
    
    \auxLargo{factorCategoria}{s: AirMiles, u:Usuario}{\R}{(millasTotales(s,u) < 10.000 \implica res = 1) \lor \\ (10.000 \leq millasTotales(s,u) < 50.000 \implica res = 1.5) \lor \\ (50.000 \leq millasTotales(s,u) < 1.000.000 \implica res = 2) \lor \\ (millasTotales (s,u) \geq 1.000.000 \implies res = 0)}

    \begin{proc}{crearSistema}{}{\Tipo{AirMiles}}
    
        \requiere{True}
    
        \aseguraLargo{res.historialDeAcumulaciones = \{\} \ \land \\ res.historialDeTransferencias = <> \ \land \\ res.historialDeCanjes \{\} \ \land \\ res.referidos = \{\} \ \land \\ res.promociones = \{"NoPromo" : (1,0)\}}
         
    \end{proc}

    \predLargo{usuarioEnSistema}{s: AirMiles, u: Usuario}{u \in usuarios(s)}
    
    \bigskip
    
    \predLargo{seAgregoUsuario}{s: AirMiles, u: Usuario}{u \in s.historialDeAcumulaciones \land s.historialDeAcumulaciones[u] = <>}

    \predLargo{lasDemasAcumulacionesNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{(\forall usu:Usuario)(usu \in claves(S_0.historialDeAcumulaciones) \implica \\ usu \in claves(s.historialDeAcumulaciones)) \ \land \\ (\forall usu:Usuario)(usu \in claves(S_0.historialDeAcumulaciones) \ \land \ usu \neq u \implica \\ s.historialDeAcumulaciones = S_0.historialDeAcumulaciones)}
    
    \bigskip
    
    \predLargo{canjesNoCambia}{s: AirMiles, S_0: AirMiles}{s.historialDeCanjes = S_0.historialDeCanjes}
    
    \bigskip
    
    \predLargo{transferenciasNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{s.historialDeTransferencias = S_0.historialDeTransferencias}
    
    \bigskip
    
    \predLargo{referidosNoCambia}{s: AirMiles, S_0: AirMiles}{s.referidos = S_0.referidos}
    
    \bigskip
    
    \predLargo{promocionesNoCambia}{s: AirMiles, S_0: AirMiles}{s.promociones = S_0.promociones}

    \begin{proc}{registrarUsuario}{\Inout s:\Tipo{AirMiles}, \In u:\Tipo{Usuario}}{}
    
        \requiere{u \geq 0 \land \neg usuarioEnSistema(s,u) \land s = S_0}
    
        \aseguraLargo{seAgregoUsuario(s,u) \land \\ (millasDisponibles(s,u) = 0 \land millasTotales(s,u) = 0) \land \\ lasDemasAcumulacionesNoCambian(s,S_0,u) \land \\ canjesNoCambia(s,S_0) \land \\  transferenciasNoCambian(s,S_0) \land \\ referidosNoCambia(s,S_0) \land \\ promocionesNoCambia(s,S_0)}
    \end{proc}
    
    \bigskip
    
    \predLargo{losDemasReferidosNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario, ref: Usuario}{(\forall usu:\Z)(usu \in claves(S_0.referidos) \land \implica usu \in claves(s.referidos))\land \\ (\forall usu:\Z)(usu \in claves(S_0.referidos) \land usu \neq u \land usu \neq ref \implica S_0.referidos[usu] = s.referidos[usu])}
    
    \bigskip
    
    
    
    \predLargo{seAgregoReferido}{s: AirMiles, S_0: AirMiles, u: Usuario, ref: Usuario}{s.referidos[u] = <500> \land \ s.referidos[ref] = S_0.referidos[ref] ++ <500> \\ \land (millasDisponibles(s,u) = 500 \land millasTotales(s,u) = 500) \land \\ (millasDisponibles(s,ref) = millasDisponibles(S_0,ref) + 500 \ \land \\ millasTotales(s,ref) = millasTotales(S_0,ref) + 500)}
    
    \bigskip
    
    
    \begin{proc}{registrarUsuarioConReferidor}{\Inout s:\Tipo{AirMiles}, \In ref:\Tipo{Usuario}, \In u:\Tipo{Usuario}}{}
    
        \requiere{u \geq 0 \land \neg usuarioEnSistema(s,u) \land usuarioEnSistema(s,ref) \land s=S_0}
    
        \aseguraLargo{seAgregoUsuario(s,u) \ \land \\ seAgregoReferido(s,S_0,u,ref) \ \land \\  lasDemasAcumulacionesNoCambian(s,S_0,u) \ \land \\ canjesNoCambia(s,S_0) \ \land \\ transferenciasNoCambian(s,S_0) \ \land \\ losDemasReferidosNoCambian(s,S_0,u,ref) \ \land \\ promocionesNoCambia(s,S_0)}
    
    \end{proc}
    
    \bigskip
    
    \begin{proc}{consultarSaldo}{\In s:\Tipo{AirMiles}, u:\Tipo{Usuario}}{\R}
    
        \requiere{usuarioEnSistema(s,u)}
    
        \asegura{res = millasDisponibles(s,u)}
         
    \end{proc}
    
    \bigskip
    
    \begin{proc}{consultarCategoria}{\In s:\Tipo{AirMiles}, u:\Tipo{Usuario}}{\Tipo{Categoria}}
    
        \requiere{usuarioEnSistema(s,u)}
    
        \asegura{res = categoria(s,u)}
    
    \end{proc}

    \predLargo{losDemasCanjesNoCambian}{s: AirMiles, u: Usuario}{(\forall usu:Usuario)(usu \in claves(S_0.historialDeCanjes) \implica usu\in claves(s.historialDeCanjes)) \land (\forall usu:Usuario)(usu \in claves(S_0.historialDeCanjes) \land usu \neq u \implica s.historialDeCanjes[usu] = S_0.historialDeCanjes[usu])}
    
    \bigskip
    
    \predLargo{acumulacionesNoCambia}{s: AirMiles, S_0: AirMiles}{s.historialDeAcumulaciones = S_0.historialDeAcumulaciones}

    \predLargo{realizarCanje}{s: AirMiles, S_0: AirMiles, u: Usuario, millas:\R}{s.historialDeCanjes[u] = s_0.historial[u] ++ <millas> \land \\ millasDisponibles(s, u) = millasDisponibles(S_0, u) - millas}
    
    \bigskip
    
    \predLargo{millasTotalesNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{millasTotales(s,u) = millasTotales(S_0,u)}
    
    \bigskip
    
    
    
    \begin{proc}{canjearMillas}{\Inout s: \Tipo{AirMiles}, u:\Tipo{Usuario}, millas:\R}{}
          
        \requiereLargo{s = S_0 \ \land \ millas > 0  \ \land usuarioEnSistema(s,u) \ \yLuego \ millas < millasDisponibles(s,u)}
          
        \aseguraLargo{realizarCanje(s,S_0,u,millas) \ \land \\ losDemasCanjesNoCambian(s,u) \ \land \\ promocionesNoCambia(s,S_0) \ \land \\ acumulacionesNoCambia(s,S_0) \ \land \\ referidosNoCambia(s,S_0) \\ land transferenciasNoCambian(s,S_0)}
          
    \end{proc}

    \begin{proc}{rankingMillas}{\In N: \Z, \In s: \Tipo{AirMiles}}{\Tipo{seq}\langle\Z\rangle}{}

        \requiereLargo{N > 0}

        \aseguraLargo{
            estaOrdenadoSegunCantidadMillasDisponibles(res,s)
            \ \land \\
            todosUsuariosEstanEnElHistorial(res,s)
            \ \land \\
            ((|s.historialDeAcumulaciones| \geq N \ \land \ |res| = N)
            \ \lor \\
            (|s.historialDeAcumulaciones| < N
            \ \land \ |res| = |s.historialDeAcumulaciones|))}
    \end{proc}

\predLargo{estaOrdenadoSegunCantidadMillasDisponibles}{u: \Tipo{seq}\langle\Z\rangle, s: \Tipo{AirMiles}}{
    (\forall i:\Z)(
        0 \leq i < |u|-1
        \ \yLuego\
        millasDisponibles(s,u[i]) \geq millasDisponibles(s,u[i+1])
    )}

\pred{todosUsuariosEstanEnElHistorial}{u: \Tipo{seq}\langle\Z\rangle, s: \Tipo{AirMiles}}{
    (\forall i:\Z)(
        0 \leq i < |u|
        \ \yLuego\
        u[i] \in s.historialDeAcumulaciones
    )}

    \begin{proc}{iniciarPromocion}
    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha},
        \In factor: \R
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\
            factor \geq 1 \land \\
            s = s_0 \land \\
            %nombre \notin s.promociones
            \neg existePromo(nombre, s_0)
        }
        \aseguraLargo{
            seAgregaPromocion(s_0, s, nombre, \struct{factor: factor, tope: {-1.0}}) \land \\
            mismosHistorialesYReferidos(s_0, s)
        }
    \end{proc}

    \bigskip



    \begin{proc}{iniciarPromocionConTope}
    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha},
        \In factor: \R,
        \In tope: \R
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\
            factor \geq 1 \land \\
            tope > 0 \land \\
            s = s_0 \land \\
            %nombre \notin s.promociones
            \neg existePromo(nombre, s_0)
        }
        \aseguraLargo{
            seAgregaPromocion(s_0, s, nombre, \struct{factor: factor, tope: tope}) \land \\
            mismosHistorialesYReferidos(s_0, s)
        }
    \end{proc}

    \bigskip



    \begin{proc}{finalizarPromocion}
    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha}
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\
            nombre \neq \text{``NoPromo''} \land \\
            nombre \in s.promociones \land \\
            s = s_0
        }
        \aseguraLargo{
            seEliminaPromocion(s_0, s, nombre) \land \\
            mismosHistorialesYReferidos(s_0, s)
        }
    \end{proc}

    \bigskip



    \predLargo{mismosHistorialesYReferidos}{s_1: \Tipo{AirMiles}, s_2: \Tipo{AirMiles}}{
        s_1.historialDeTransferencias = s_2.historialDeTransferencias \land \\
        s_1.historialDeAcumulaciones = s_2.historialDeAcumulaciones \land \\
        s_1.historialDeCanjes = s_2.historialDeCanjes \land \\
        s_1.referidos = s_2.referidos
    }

    \bigskip



    \predLargo{seAgregaPromocion}{s_0: \Tipo{AirMiles}, s: \Tipo{AirMiles}, nombre: \seq{\cha}, datosPromo: \Tipo{ParametrosPromo}}{
        \norm{s.promociones} = \norm{s_0.promociones} + 1 \land \\
        nombre \in s.promociones \land \\
        s.promociones[nombre] = datosPromo \land \\
        (\forall r: \seq{\cha})(r \in s_0.promociones \implicaLuego \\
        \hspace*{2em} (r \in s.promociones \yLuego s.promociones[r] = s_0.promociones[r]))
    }

    \bigskip



    \predLargo{seEliminaPromocion}{s_0: \Tipo{AirMiles}, s: \Tipo{AirMiles}, nombre: \seq{\cha}}{
        \norm{s_0.promociones} = \norm{s.promociones} + 1 \land \\
        nombre \notin s.promociones \land \\
        (\forall r: \seq{\cha})(r \in s.promociones \implicaLuego \\
        \hspace*{2em} (r \in s_0.promociones \yLuego s_0.promociones[r] = s.promociones[r]))
    }

\bigskip

Pequeño comentario sobre como funciona el proc acumularMillas (pues se ve horrible en el documento): este proc simplemente garantiza que solo se modifica la cantidad de vuelos de un usuario en especifico y que modifica el historial de promociones condicionalmente, al final del proc, se utiliza el pred miIfThenElse (que se redefinio para no tener que abusar del ifThenElse "nativo" del lenguaje) y demostrar que no estamos usando el ifThenElse como si estuvieramos programando.

    la estructura de la condicion es equivalente a
    \begin{verbatim}
    if(LaPromoNoTieneTope)
        then casoNoTope
    else if(elTopeDeLaPromoEsIgualALoQueElUsuarioVaAVolar)
        then casoGastaElTopeDeLaPromo
        else casoNoGastaElTopeDeLaPromo
    \end{verbatim}

    \bigskip

    \auxLargo{sumaDeTransferenciasRecibidas}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.historialDeTransferencias|-1}
        \operatorname{IfThenElse}\big(
            s.historialDeTransferencias[i].destino = u, s.historialDeTransferencias[i].cantidad,0\big)}
    
    \bigskip
    
    \auxLargo{sumaDeTransferenciasEnviadas}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.historialDeTransferencias|-1}
        \operatorname{IfThenElse}\big(
        s.historialDeTransferencias[i].origen = u, s.historialDeTransferencias[i].cantidad,0\big)}
    
    \bigskip
    
    \aux{sumaReferidos}{s: AirMiles, u: Usuario}{\R}{\sum_{i=0}^{|s.referidos[u]|-1}{s.referidos[u][i]}}

    \bigskip

    \auxLargo{sumaDeCanjes}{u: Usuario, s: \Tipo{AirMiles}}{\R}{\sum_{i=0}^{|s.historialDeCanjes[u]|-1}
    s.historialDeCanjes[u][i]}
    
    \bigskip

    \begin{proc}{crearSistema}{}{\Tipo{AirMiles}}
     
        \requiere{True}
    
        \aseguraLargo{res.historialDeAcumulaciones = \{\} \ \land \\ res.historialDeTransferencias = <> \ \land \\ res.historialDeCanjes = \{\} \ \land \\ res.referidos = \{\} \ \land \\ res.promociones = \{"NoPromo" : \struct{1,-1}\}}

    \predLargo{seAgregoUsuario}{s: AirMiles, u: Usuario}{u \in s.historialDeAcumulaciones \land s.historialDeAcumulaciones[u] = <> \ \land \\ u \in s.historialDeCanjes \land s.historialDeCanjes[u] = <> \ land \\ u \in s.referidos \land s.referidos[u] = <>}
    
    \bigskip
    
    \predLargo{lasDemasAcumulacionesNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{(\forall usu:Usuario)(usu \in claves(S_0.historialDeAcumulaciones) \implica \\ usu \in claves(s.historialDeAcumulaciones)) \ \land \\ (\forall usu:Usuario)((usu \in claves(S_0.historialDeAcumulaciones) \ \land \ usu \neq u) \implicaLuego \\ s.historialDeAcumulaciones[usu] = S_0.historialDeAcumulaciones[usu])}
    
    \bigskip
    
    \predLargo{losDemasCanjesNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{(\forall usu:Usuario)(usu \in claves(S_0.historialDeCanjes) \implica usu\in claves(s.historialDeCanjes)) \land (\forall usu:Usuario)((usu \in claves(S_0.historialDeCanjes) \land usu \neq u) \implicaLuego s.historialDeCanjes[usu] = S_0.historialDeCanjes[usu])}
    
    \bigskip
    
    \predLargo{losDemasReferidosNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario}{(\forall usu: Usuario)(usu \in claves(S_0.referidos) \implica usu \in claves(s.referidos)) \land (\forall usu: Usuario)((usu \in claves(S_0.referidos) \land usu \neq u) \implicaLuego s.referidos[usu] = S_0.referidos[usu])}
    
    \predLargo{transferenciasNoCambian}{s: AirMiles, S_0: AirMiles}{s.historialDeTransferencias = S_0.historialDeTransferencias}
    
    \bigskip
    
    \predLargo{promocionesNoCambia}{s: AirMiles, S_0: AirMiles}{s.promociones = S_0.promociones}

    \begin{proc}{registrarUsuario}{\Inout s:\Tipo{AirMiles}, \In u:\Tipo{Usuario}}{}
    
        \requiere{u > 0 \land \neg usuarioEnSistema(s,u) \land s = S_0}
    
        \aseguraLargo{seAgregoUsuario(s,u) \ \land \\ lasDemasAcumulacionesNoCambian(s,S_0,u) \land \\ losDemasCanjesNoCambian(s,S_0,u) \land \\ losDemasReferidosNoCambian(s,S_0,u)  \land \\ transferenciasNoCambian(s,S_0) \land \\ promocionesNoCambia(s,S_0)}
    \end{proc}
    
    \bigskip
    
    
    \predLargo{losOtrosReferidosNoCambian}{s: AirMiles, S_0: AirMiles, u: Usuario, ref: Usuario}{(\forall usu:\Z)(usu \in claves(S_0.referidos) \land \implica usu \in claves(s.referidos))\land \\ (\forall usu:\Z)((usu \in claves(S_0.referidos) \land usu \neq u \land usu \neq ref) \implicaLuego S_0.referidos[usu] = s.referidos[usu])}

    \predLargo{seAgregoReferido}{s: AirMiles, S_0: AirMiles, u: Usuario, ref: Usuario}{s.referidos[u] = S_0.referidos[u] ++ <500> \ \land \ s.referidos[ref] = S_0.referidos[ref] ++ <500>}

    \end{proc}

    \begin{proc}{rankingMillas}{\In N: \Z, \In s: \Tipo{AirMiles}}{\seq{Usuario}}{}

        \requiereLargo{N > 0}

        \aseguraLargo{
            estaOrdenadoSegunCantidadMillasDisponibles(res,s)
            \ \land \\
            todosUsuariosEstanEnElHistorial(res,s)
            \ \land \\ resTieneTamañoMenorOIgualQueN(res, s, N) \ land \\ todosUsuariosDistintos(res) \ land \\ todosLosNoSeleccionadosTienenMenosOMismasMillas(res,s)
            }
    \end{proc}

\predLargo{estaOrdenadoSegunCantidadMillasDisponibles}{u: \Tipo{seq}\langle\Z\rangle, s: \Tipo{AirMiles}}{
    (\forall i:\Z)(
        0 \leq i < |u|-1
        \ \yLuego\
        millasDisponibles(s,u[i]) \geq millasDisponibles(s,u[i+1])
    )}
    
\bigskip

\predLargo{todosUsuariosEstanEnElHistorial}{res: \seq{Usuario}, s: \Tipo{AirMiles}}{
    (\forall i:\Z)(
        0 \leq i < |u|
        \ \implicaLuego \
        res[i] \in usuarios(s)
    )}
    
\bigskip

\predLargo{resTieneTamañoMenorOIgualQueN}{res:\seq{Usuario}, s: AirMiles, N: \Z}{(N \leq |usuarios(s)| \ \land \ |res| = N)
            \ \lor \\
            (|usuarios(s)| < N
            \ \land \ |res| = |s.historialDeAcumulaciones|)}

    \bigskip
    
    \predLargo{todosUsuariosDistintos}{res:\seq{Usuario}}{
    (\forall i:\Z)(
        0 \leq i < |res|
        \ \rightarrow_L\
        (\forall j:\Z)(
            0 \leq j < |res|
            \ \land\
            i \neq j
            \ \implicaLuego \
            res[i] \neq res[j]
        )
    )
}

\bigskip

\predLargo{todosLosNoSeleccionadosTienenMenosOMismasMillas}
{res:\seq{Usuario},s:\Tipo{AirMiles}}{
    (\forall u:\Z)(
        u \in usuarios(s)
        \ \land\
        u \notin res
        \ \rightarrow_L\
        millasDisponibles(s,u)
        \leq
        millasDisponibles(s,res[|res|-1])
    )
}

    \begin{proc}{iniciarPromocion}
    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha},
        \In factor: \R
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\
            factor \geq 1 \land \\
            s = s_0 \land \\
            %nombre \notin s.promociones
            \neg existePromo(nombre, s_0)
        }
        \aseguraLargo{
            seAgregaPromocion(s_0, s, nombre, \struct{factor: factor, tope: {-1.0}}) \land \\
            acumulacionesNoCambia(s_0, s) \land \\
            canjesNoCambia(s_0, s) \land \\
            transferenciasNoCambian(s_0, s) \land \\
            referidosNoCambia(s_0, s)
        }
    \end{proc}

    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha},
        \In factor: \R,
        \In tope: \R
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\

            acumulacionesNoCambia(s_0, s) \land \\
            canjesNoCambia(s_0, s) \land \\
            transferenciasNoCambian(s_0, s) \land \\
            referidosNoCambia(s_0, s)
        }

    \begin{proc}{finalizarPromocion}
    {
        \Inout s: \Tipo{AirMiles},
        \In nombre: \seq{\cha}
    }{}
        \requiereLargo{
            \norm{nombre} > 0 \land \\
            nombre \neq \text{``NoPromo''} \land \\
            nombre \in s.promociones \land \\
            s = s_0
        }
        \aseguraLargo{
            seEliminaPromocion(s_0, s, nombre) \land \\
            acumulacionesNoCambia(s_0, s) \land \\
            canjesNoCambia(s_0, s) \land \\
            transferenciasNoCambian(s_0, s) \land \\
            referidosNoCambia(s_0, s)
        }
    \end{proc}

% --- Existe al menos una transferencia entre a y b ---
\predLargo{sonTransferidores}
    {a: Usuario, b: Usuario, s: AirMiles}
{
    (\exists i:\Z)(
        0 \leq i < |s.historialDeTransferencias|
        \ \land \\
        (
            (s.historialDeTransferencias[i].origen = a
            \land
            s.historialDeTransferencias[i].destino = b)
            \ \lor \\
            (s.historialDeTransferencias[i].origen = b
            \land
            s.historialDeTransferencias[i].destino = a)
        )
    )
}

\bigskip


% --- a solamente le transfiere a b ---
\predLargo{usuarioAsoloLeTransfiereExclusivamenteAlUsuarioB}
    {a: Usuario, b: Usuario, s: AirMiles}
{
    (\forall i:\Z)(
        0 \leq i < |s.historialDeTransferencias|
        \ \implicaLuego \\
        (
            s.historialDeTransferencias[i].origen = a
            \implica
            s.historialDeTransferencias[i].destino = b
        )
    )
}

\bigskip


% --- b solamente le transfiere a a ---
% Se obtiene reutilizando el predicado anterior.
\predLargo{sonTransferidoresExclusivos}
    {a: Usuario, b: Usuario, s: AirMiles}
{
    a \neq b
    \ \land \\
    sonTransferidores(a,b,s)
    \ \land \\
    usuarioAsoloLeTransfiereExclusivamenteAlUsuarioB(a,b,s)
    \ \land \\
    usuarioAsoloLeTransfiereExclusivamenteAlUsuarioB(b,a,s)
}

\bigskip


% --- Todo par que aparece en res es exclusivo ---
\predLargo{todosLosParesEnResSonExclusivos}
    {res:\seq{\tupla{Usuario, Usuario}},
     s:\Tipo{AirMiles}}
{
    (\forall i:\Z)(
        0 \leq i < |res|
        \ \implicaLuego \
        sonTransferidoresExclusivos(
            res[i].0,
            res[i].1,
            s
        )
    )
}

\bigskip


% --- Todo par exclusivo aparece en res ---
\predLargo{enResEstanTodosLosExclusivos}
    {res:\seq{\tupla{Usuario, Usuario}},
     s:\Tipo{AirMiles}}
{
    (\forall a,b:Usuario)(
        (
            usuarioEnSistema(s,a)
            \land
            usuarioEnSistema(s,b)
        )
        \ \implicaLuego \\
        (
            sonTransferidoresExclusivos(a,b,s)
            \implica
            \tupla{a,b} \in res
        )
    )
}

\bigskip


% --- No se repite exactamente el mismo par ---
\predLargo{noHayParesRepetidos}
    {res:\seq{\tupla{Usuario, Usuario}}}
{
    (\forall i,j:\Z)(
        (
            0 \leq i < |res|
            \land
            0 \leq j < |res|
            \land
            i \neq j
        )
        \ \implicaLuego
        res[i] \neq res[j]
    )
}

\bigskip


% --- No aparecen ambos órdenes del mismo par ---
\predLargo{noSeInviertenEnRes}
    {res:\seq{\tupla{Usuario, Usuario}}}
